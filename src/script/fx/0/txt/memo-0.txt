#!/bin/bash

# 1. 各種設定
AQTKPI="/home/pi/root/sys/env/tool/aquestalkpi/1.20/AquesTalkPi"
TEXT="アクエストークパイの音質改善テストです。サシスセソ、ツクシ。"

# 一時ファイルの保存先
ORIGINAL_WAV="/tmp/aq_original.wav"
EQ_WAV="/tmp/aq_eq.wav"

# 2. 音声ファイルの生成
# オリジナルの生データ（8000Hz）を作成
"$AQTKPI" "$TEXT" > "$ORIGINAL_WAV"

# 【アップサンプリング経由の処理】
# -r 44100 : 内部で一度44.1kHzに引き上げる
# highpass / equalizer : 広い帯域の中で歪みなく綺麗に音を補正する
# -r 8000  : 補正した綺麗な状態のまま、ターゲットの8000Hzにダウンサンプリングして戻す
sox "$ORIGINAL_WAV" -r 44100 -t wav - highpass 150 equalizer 3000 1.0q 6 equalizer 4000 1.2q -4 | \
sox -t wav - -r 8000 "$EQ_WAV"

# 3. 連続再生して聴き比べ
echo "========================================"
echo "◆ 1番目：【オリジナル】（加工なし 8000Hz）"
echo "========================================"
aplay "$ORIGINAL_WAV"

sleep 1

echo "========================================"
echo "◆ 2番目：【アップサンプリング改善版】（音量維持・8000Hz）"
echo "========================================"
aplay "$EQ_WAV"

# 4. 後片付け
rm -f "$ORIGINAL_WAV" "$EQ_WAV"

exit 0
